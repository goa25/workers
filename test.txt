import React, { useState } from 'react';
import { Upload, CheckCircle, Clock, XCircle, AlertTriangle, ChevronDown, ChevronUp, FileText, Leaf, Calendar, X, Send } from 'lucide-react';

type EvidenceStatus = 'pending' | 'verified' | 'rejected' | 'flagged';

interface Activity {
  id: string; pledgeId: string; pledgeTitle: string; category: string;
  month: string; submittedAt: string; status: EvidenceStatus;
  co2Saved: number; notes: string; evidenceFile?: string;
}

const enrolledPledges = [
  { id: 'public-transport', title: 'Public Transport', category: 'Transportation', icon: 'Bus',
    fields: [
      { label: 'Days used public transport', type: 'number', placeholder: '0', unit: 'days' },
      { label: 'Total km traveled', type: 'number', placeholder: '0', unit: 'km' },
      { label: 'Mode used', type: 'select', options: ['Metro/Subway', 'Bus', 'Train', 'Mixed'] },
    ]
  },
  { id: 'plant-based-meals', title: 'Plant-Based Meals', category: 'Food', icon: 'Salad',
    fields: [
      { label: 'Plant-based meals this month', type: 'number', placeholder: '0', unit: 'meals' },
      { label: 'Total meals this month', type: 'number', placeholder: '0', unit: 'meals' },
    ]
  },
  { id: 'tree-plantation', title: 'Tree Plantation', category: 'Nature', icon: 'Tree',
    fields: [
      { label: 'Trees planted this month', type: 'number', placeholder: '0', unit: 'trees' },
      { label: 'Survival rate', type: 'number', placeholder: '0', unit: '%' },
    ]
  },
  { id: 'led-replacement', title: 'LED Replacement', category: 'Energy', icon: 'Bulb',
    fields: [
      { label: 'Total LED bulbs active', type: 'number', placeholder: '0', unit: 'bulbs' },
      { label: 'Estimated monthly savings', type: 'number', placeholder: '0', unit: 'kWh' },
    ]
  },
];

const pastActivities: Activity[] = [
  { id: '1', pledgeId: 'public-transport', pledgeTitle: 'Public Transport', category: 'Transportation',
    month: 'May 2026', submittedAt: '2026-06-01', status: 'verified', co2Saved: 48.2,
    notes: 'Used metro 22 days, covered 330km total', evidenceFile: 'metro_pass_may26.jpg' },
  { id: '2', pledgeId: 'plant-based-meals', pledgeTitle: 'Plant-Based Meals', category: 'Food',
    month: 'May 2026', submittedAt: '2026-06-01', status: 'pending', co2Saved: 32.1,
    notes: '18 plant-based out of 90 meals', evidenceFile: 'meal_log_may26.pdf' },
  { id: '3', pledgeId: 'tree-plantation', pledgeTitle: 'Tree Plantation', category: 'Nature',
    month: 'April 2026', submittedAt: '2026-05-02', status: 'verified', co2Saved: 5.0,
    notes: 'Planted 5 native saplings in community park', evidenceFile: 'tree_photos_apr26.jpg' },
  { id: '4', pledgeId: 'led-replacement', pledgeTitle: 'LED Replacement', category: 'Energy',
    month: 'April 2026', submittedAt: '2026-05-02', status: 'rejected', co2Saved: 0,
    notes: 'Evidence photo unclear, please resubmit' },
];

const months = ['June 2026', 'May 2026', 'April 2026', 'March 2026'];

const statusConfig: Record<EvidenceStatus, { label: string; color: string }> = {
  pending: { label: 'Pending Review', color: 'text-amber-600 bg-amber-50 border-amber-200' },
  verified: { label: 'Verified', color: 'text-green-600 bg-green-50 border-green-200' },
  rejected: { label: 'Rejected', color: 'text-red-600 bg-red-50 border-red-200' },
  flagged: { label: 'Flagged', color: 'text-orange-600 bg-orange-50 border-orange-200' },
};

export default function ActivityReport() {
  const [activeTab, setActiveTab] = useState<'submit' | 'history'>('submit');
  const [selectedPledge, setSelectedPledge] = useState('');
  const [selectedMonth, setSelectedMonth] = useState('June 2026');
  const [formValues, setFormValues] = useState<Record<string, string>>({});
  const [notes, setNotes] = useState('');
  const [uploadedFile, setUploadedFile] = useState<File | null>(null);
  const [dragOver, setDragOver] = useState(false);
  const [submitted, setSubmitted] = useState(false);
  const [expandedActivity, setExpandedActivity] = useState<string | null>(null);
  const [filterStatus, setFilterStatus] = useState<EvidenceStatus | 'all'>('all');

  const currentPledge = enrolledPledges.find(p => p.id === selectedPledge);

  const handleFileChange = (e: React.ChangeEvent<HTMLInputElement>) => {
    const file = e.target.files?.[0];
    if (!file) return;
    if (file.size > 10 * 1024 * 1024) { alert('File size must be under 10MB'); return; }
    if (!['image/jpeg', 'image/png', 'application/pdf'].includes(file.type)) {
      alert('Only JPG, PNG, and PDF files are accepted'); return;
    }
    setUploadedFile(file);
  };

  const handleDrop = (e: React.DragEvent) => {
    e.preventDefault(); setDragOver(false);
    const file = e.dataTransfer.files[0];
    if (!file) return;
    if (file.size > 10 * 1024 * 1024) { alert('File size must be under 10MB'); return; }
    if (!['image/jpeg', 'image/png', 'application/pdf'].includes(file.type)) {
      alert('Only JPG, PNG, and PDF files are accepted'); return;
    }
    setUploadedFile(file);
  };

  const handleSubmit = () => {
    if (!selectedPledge || !uploadedFile) return;
    setSubmitted(true);
    setTimeout(() => { setSubmitted(false); setSelectedPledge(''); setFormValues({}); setNotes(''); setUploadedFile(null); }, 3000);
  };

  const filteredActivities = filterStatus === 'all' ? pastActivities : pastActivities.filter(a => a.status === filterStatus);

  return (
    <div className="min-h-screen bg-slate-50">
      <div className="bg-white border-b border-slate-200">
        <div className="max-w-5xl mx-auto px-4 sm:px-6 lg:px-8 py-8">
          <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
            <div>
              <h1 className="text-3xl font-bold text-slate-900 font-display">Activity Reporting</h1>
              <p className="mt-1 text-slate-500">Submit monthly evidence for your pledges to track verified impact</p>
            </div>
            <div className="flex items-center gap-2 text-sm bg-brand-50 text-brand-700 px-4 py-2 rounded-full font-medium">
              <Leaf size={16} />
              <span>Total verified: 53.2 kg CO2</span>
            </div>
          </div>
          <div className="mt-6 flex gap-1 border-b border-slate-200">
            {(['submit', 'history'] as const).map(tab => (
              <button key={tab} onClick={() => setActiveTab(tab)}
                className={`px-5 py-3 text-sm font-medium capitalize transition-colors border-b-2 -mb-px ${
                  activeTab === tab ? 'border-brand-600 text-brand-600' : 'border-transparent text-slate-500 hover:text-slate-700'
                }`}>
                {tab === 'submit' ? 'Submit Activity' : 'Submission History'}
              </button>
            ))}
          </div>
        </div>
      </div>
      <div className="max-w-5xl mx-auto px-4 sm:px-6 lg:px-8 py-8">
        {submitted && (
          <div className="mb-6 flex items-center gap-3 bg-brand-600 text-white px-5 py-4 rounded-xl shadow-lg">
            <CheckCircle size={20} />
            <div>
              <p className="font-semibold">Activity submitted successfully!</p>
              <p className="text-brand-100 text-sm">Your evidence is under review. You will be notified when verified.</p>
            </div>
          </div>
        )}
        {activeTab === 'submit' && (
          <div className="grid grid-cols-1 lg:grid-cols-3 gap-8">
            <div className="lg:col-span-2 space-y-6">
              <div className="bg-white rounded-2xl border border-slate-200 p-6 shadow-sm">
                <h2 className="font-semibold text-slate-900 mb-4 flex items-center gap-2">
                  <span className="w-6 h-6 rounded-full bg-brand-600 text-white text-xs flex items-center justify-center font-bold">1</span>
                  Select Pledge and Reporting Period
                </h2>
                <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
                  <div>
                    <label className="block text-sm font-medium text-slate-700 mb-2">Pledge</label>
                    <select value={selectedPledge} onChange={e => { setSelectedPledge(e.target.value); setFormValues({}); }}
                      className="w-full border border-slate-200 rounded-xl px-4 py-3 text-sm focus:outline-none focus:ring-2 focus:ring-brand-500 bg-white">
                      <option value="">Choose an enrolled pledge...</option>
                      {enrolledPledges.map(p => <option key={p.id} value={p.id}>{p.title}</option>)}
                    </select>
                  </div>
                  <div>
                    <label className="block text-sm font-medium text-slate-700 mb-2">Reporting Month</label>
                    <select value={selectedMonth} onChange={e => setSelectedMonth(e.target.value)}
                      className="w-full border border-slate-200 rounded-xl px-4 py-3 text-sm focus:outline-none focus:ring-2 focus:ring-brand-500 bg-white">
                      {months.map(m => <option key={m} value={m}>{m}</option>)}
                    </select>
                  </div>
                </div>
              </div>
              {currentPledge && (
                <div className="bg-white rounded-2xl border border-slate-200 p-6 shadow-sm">
                  <h2 className="font-semibold text-slate-900 mb-4 flex items-center gap-2">
                    <span className="w-6 h-6 rounded-full bg-brand-600 text-white text-xs flex items-center justify-center font-bold">2</span>
                    Activity Data for {currentPledge.title}
                  </h2>
                  <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
                    {currentPledge.fields.map((field, idx) => (
                      <div key={idx}>
                        <label className="block text-sm font-medium text-slate-700 mb-1">
                          {field.label}
                          {field.unit && <span className="text-slate-400 font-normal ml-1">({field.unit})</span>}
                        </label>
                        {field.type === 'select' ? (
                          <select value={formValues[String(idx)] || ''} onChange={e => setFormValues(prev => ({ ...prev, [String(idx)]: e.target.value }))}
                            className="w-full border border-slate-200 rounded-xl px-4 py-3 text-sm focus:outline-none focus:ring-2 focus:ring-brand-500 bg-white">
                            <option value="">Select option</option>
                            {field.options?.map(o => <option key={o} value={o}>{o}</option>)}
                          </select>
                        ) : (
                          <input type={field.type} placeholder={field.placeholder} value={formValues[String(idx)] || ''}
                            onChange={e => setFormValues(prev => ({ ...prev, [String(idx)]: e.target.value }))}
                            className="w-full border border-slate-200 rounded-xl px-4 py-3 text-sm focus:outline-none focus:ring-2 focus:ring-brand-500 bg-white" />
                        )}
                      </div>
                    ))}
                  </div>
                  <div className="mt-4">
                    <label className="block text-sm font-medium text-slate-700 mb-2">Additional Notes (optional)</label>
                    <textarea value={notes} onChange={e => setNotes(e.target.value)}
                      placeholder="Add context about your activity this month..." rows={3}
                      className="w-full border border-slate-200 rounded-xl px-4 py-3 text-sm focus:outline-none focus:ring-2 focus:ring-brand-500 bg-white resize-none" />
                  </div>
                </div>
              )}
              {currentPledge && (
                <div className="bg-white rounded-2xl border border-slate-200 p-6 shadow-sm">
                  <h2 className="font-semibold text-slate-900 mb-2 flex items-center gap-2">
                    <span className="w-6 h-6 rounded-full bg-brand-600 text-white text-xs flex items-center justify-center font-bold">3</span>
                    Upload Evidence
                  </h2>
                  <p className="text-sm text-slate-500 mb-4">Accepted: JPG, PNG, PDF (max 10MB)</p>
                  {!uploadedFile ? (
                    <div onDragOver={e => { e.preventDefault(); setDragOver(true); }} onDragLeave={() => setDragOver(false)} onDrop={handleDrop}
                      className={`border-2 border-dashed rounded-2xl p-10 text-center transition-colors ${dragOver ? 'border-brand-400 bg-brand-50' : 'border-slate-200 hover:border-brand-300'}`}>
                      <Upload size={36} className="mx-auto text-slate-300 mb-3" />
                      <p className="text-slate-500 text-sm font-medium">Drag and drop your file here</p>
                      <p className="text-slate-400 text-xs mt-1 mb-4">or click to browse</p>
                      <label className="cursor-pointer bg-brand-600 hover:bg-brand-700 text-white text-sm font-semibold px-5 py-2.5 rounded-xl transition-colors">
                        Choose File <input type="file" className="hidden" accept=".jpg,.jpeg,.png,.pdf" onChange={handleFileChange} />
                      </label>
                    </div>
                  ) : (
                    <div className="flex items-center gap-4 p-4 bg-brand-50 rounded-xl border border-brand-200">
                      <FileText size={20} className="text-brand-600" />
                      <div className="flex-1 min-w-0">
                        <p className="font-medium text-slate-800 text-sm truncate">{uploadedFile.name}</p>
                        <p className="text-slate-500 text-xs">{(uploadedFile.size / 1024).toFixed(0)} KB</p>
                      </div>
                      <button onClick={() => setUploadedFile(null)} className="text-slate-400 hover:text-red-500"><X size={18} /></button>
                    </div>
                  )}
                </div>
              )}
              {currentPledge ? (
                <button onClick={handleSubmit} disabled={!uploadedFile}
                  className={`w-full flex items-center justify-center gap-2 py-4 rounded-2xl font-semibold text-base transition-all ${
                    uploadedFile ? 'bg-brand-600 hover:bg-brand-700 text-white shadow-lg' : 'bg-slate-200 text-slate-400 cursor-not-allowed'
                  }`}>
                  <Send size={18} /> Submit Activity Report
                </button>
              ) : (
                <div className="bg-white rounded-2xl border border-slate-200 p-12 text-center shadow-sm">
                  <Leaf size={40} className="mx-auto text-slate-200 mb-4" />
                  <p className="text-slate-500 font-medium">Select a pledge above to start reporting</p>
                </div>
              )}
            </div>
            <div className="space-y-4">
              <div className="bg-white rounded-2xl border border-slate-200 p-5 shadow-sm">
                <h3 className="font-semibold text-slate-900 mb-3 flex items-center gap-2">
                  <Calendar size={16} className="text-brand-600" /> Reporting Schedule
                </h3>
                <ul className="space-y-2 text-sm text-slate-600">
                  <li className="flex items-start gap-2"><CheckCircle size={14} className="text-brand-500 mt-0.5" /> Reports due by the 5th of each month</li>
                  <li className="flex items-start gap-2"><CheckCircle size={14} className="text-brand-500 mt-0.5" /> Covers the previous calendar month</li>
                  <li className="flex items-start gap-2"><CheckCircle size={14} className="text-brand-500 mt-0.5" /> Late submissions accepted up to the 10th</li>
                </ul>
              </div>
              <div className="bg-white rounded-2xl border border-slate-200 p-5 shadow-sm">
                <h3 className="font-semibold text-slate-900 mb-3">Evidence Guidelines</h3>
                <ul className="space-y-2 text-sm text-slate-600">
                  <li className="flex items-start gap-2"><CheckCircle size={14} className="text-green-500 mt-0.5" /> Transit passes, tickets, or app screenshots</li>
                  <li className="flex items-start gap-2"><CheckCircle size={14} className="text-green-500 mt-0.5" /> Utility bills showing reduction</li>
                  <li className="flex items-start gap-2"><CheckCircle size={14} className="text-green-500 mt-0.5" /> Photos with visible date metadata</li>
                  <li className="flex items-start gap-2"><XCircle size={14} className="text-red-400 mt-0.5" /> Screenshots must show date and time</li>
                  <li className="flex items-start gap-2"><XCircle size={14} className="text-red-400 mt-0.5" /> Edited or manipulated images rejected</li>
                </ul>
              </div>
              <div className="bg-amber-50 rounded-2xl border border-amber-200 p-5">
                <h3 className="font-semibold text-amber-800 mb-2 flex items-center gap-2"><AlertTriangle size={14} /> Verification Policy</h3>
                <p className="text-sm text-amber-700">Evidence is reviewed by IISD-certified verifiers within 5 business days. Verified reports count toward leaderboard rankings and certificate eligibility.</p>
              </div>
            </div>
          </div>
        )}
        {activeTab === 'history' && (
          <div>
            <div className="flex gap-2 mb-6 flex-wrap">
              {(['all', 'pending', 'verified', 'rejected', 'flagged'] as const).map(status => (
                <button key={status} onClick={() => setFilterStatus(status)}
                  className={`px-4 py-2 rounded-full text-sm font-medium transition-colors capitalize ${
                    filterStatus === status ? 'bg-slate-800 text-white' : 'bg-white border border-slate-200 text-slate-600 hover:bg-slate-50'
                  }`}>{status}</button>
              ))}
            </div>
            <div className="space-y-4">
              {filteredActivities.map(activity => {
                const sc = statusConfig[activity.status];
                const isExpanded = expandedActivity === activity.id;
                return (
                  <div key={activity.id} className="bg-white rounded-2xl border border-slate-200 shadow-sm overflow-hidden">
                    <div className="flex items-center gap-4 p-5 cursor-pointer hover:bg-slate-50" onClick={() => setExpandedActivity(isExpanded ? null : activity.id)}>
                      <div className="w-10 h-10 rounded-xl bg-slate-100 flex items-center justify-center flex-shrink-0">
                        <Leaf size={20} className="text-brand-600" />
                      </div>
                      <div className="flex-1 min-w-0">
                        <div className="flex items-center gap-2">
                          <h3 className="font-semibold text-slate-900">{activity.pledgeTitle}</h3>
                          <span className="text-slate-400 text-sm">·</span>
                          <span className="text-sm text-slate-500">{activity.month}</span>
                        </div>
                        <p className="text-sm text-slate-500 truncate">{activity.notes}</p>
                      </div>
                      <div className="flex items-center gap-3">
                        {activity.co2Saved > 0 && <div className="text-right hidden sm:block"><p className="text-sm font-bold text-brand-700">+{activity.co2Saved} kg</p><p className="text-xs text-slate-400">CO2</p></div>}
                        <span className={`text-xs font-semibold px-3 py-1.5 rounded-full border ${sc.color}`}>{sc.label}</span>
                        {isExpanded ? <ChevronUp size={16} className="text-slate-400" /> : <ChevronDown size={16} className="text-slate-400" />}
                      </div>
                    </div>
                    {isExpanded && (
                      <div className="border-t border-slate-100 p-5 bg-slate-50">
                        <div className="grid grid-cols-3 gap-4 mb-4">
                          <div><p className="text-xs text-slate-500 mb-1">Submitted</p><p className="text-sm font-medium">{activity.submittedAt}</p></div>
                          <div><p className="text-xs text-slate-500 mb-1">Category</p><p className="text-sm font-medium">{activity.category}</p></div>
                          <div><p className="text-xs text-slate-500 mb-1">CO2 Saved</p><p className="text-sm font-bold text-brand-700">{activity.co2Saved} kg</p></div>
                        </div>
                        {activity.evidenceFile && (
                          <div className="flex items-center gap-3 p-3 bg-white rounded-xl border border-slate-200">
                            <FileText size={16} className="text-slate-400" />
                            <span className="text-sm flex-1">{activity.evidenceFile}</span>
                            <button className="text-brand-600 text-sm font-medium hover:underline">View</button>
                          </div>
                        )}
                        {activity.status === 'rejected' && (
                          <div className="mt-3 p-3 bg-red-50 rounded-xl border border-red-200">
                            <p className="text-sm text-red-700">Rejection reason: {activity.notes}</p>
                            <button className="mt-2 text-sm font-semibold text-red-600 hover:underline">Resubmit evidence</button>
                          </div>
                        )}
                      </div>
                    )}
                  </div>
                );
              })}
              {filteredActivities.length === 0 && (
                <div className="text-center py-16"><FileText size={48} className="mx-auto text-slate-200 mb-4" /><p className="text-slate-500">No submissions with this status yet.</p></div>
              )}
            </div>
          </div>
        )}
      </div>
    </div>
  );
}
